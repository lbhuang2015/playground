import os
import subprocess
import requests

def fetch_vault_secrets():
    # 模拟从 Vault 获取数据
    url = "https://your-vault-server:8200/v1/secret/data/myapp"
    headers = {"X-Vault-Token": "your_vault_token_here"}
    
    response = requests.get(url, headers=headers).json()
    secrets = response['data']['data']
    return secrets['username'], secrets['password']

def main():
    username, password = fetch_vault_secrets()
    
    # 获取当前的系统环境变量字典
    env = os.environ.copy()
    
    # 将从 Vault 拿到的秘密注入到这个字典中
    env["db_username"] = username
    env["db_password"] = password
    
    # 启动 Java 子进程，并将带有新环境变量的 env 传给它
    # 这样密码只存在于 Python 和 Java 的进程内存中，绝不落盘，ps 也看不到
    subprocess.run(["java", "-jar", "my-springboot-app.jar"], env=env)

if __name__ == "__main__":
    main()


import org.apache.spark.api.java.JavaPairRDD;
import scala.Tuple2;
import scala.Tuple3;
import java.util.ArrayList;
import java.util.List;
import java.util.concurrent.ThreadLocalRandom;

public class SparkJoinSkewUtils {

    /**
     * 对左表和右表进行加盐与膨胀处理，以解决 Join 倾斜
     * @param leftRDD 左表 JavaPairRDD<String, Tuple3>
     * @param rightRDD 右表 JavaPairRDD<String, Tuple3>
     * @param saltRange 盐值范围（建议设为核心数的 1-2 倍，如 480）
     * @return 加盐后的左表与右表
     */
    public static <A, B, C> Tuple2<JavaPairRDD<String, Tuple3<A, B, C>>, JavaPairRDD<String, Tuple3<A, B, C>>> 
    prepareSaltedJoin(
            JavaPairRDD<String, Tuple3<A, B, C>> leftRDD, 
            JavaPairRDD<String, Tuple3<A, B, C>> rightRDD, 
            int saltRange) {

        // 1. 左表随机加盐：给每个 Key 加上随机后缀 (e.g., "key_5")
        JavaPairRDD<String, Tuple3<A, B, C>> saltedLeft = leftRDD.mapToPair(tuple -> {
            int salt = ThreadLocalRandom.current().nextInt(saltRange);
            return new Tuple2<>(tuple._1() + "_" + salt, tuple._2());
        });

        // 2. 右表全量膨胀：将右表的每一条记录复制 saltRange 份，涵盖所有可能的盐值
        JavaPairRDD<String, Tuple3<A, B, C>> expandedRight = rightRDD.flatMapToPair(tuple -> {
            List<Tuple2<String, Tuple3<A, B, C>>> expandedList = new ArrayList<>();
            for (int i = 0; i < saltRange; i++) {
                expandedList.add(new Tuple2<>(tuple._1() + "_" + i, tuple._2()));
            }
            return expandedList.iterator();
        });

        return new Tuple2<>(saltedLeft, expandedRight);
    }
}
